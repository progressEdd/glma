# Plan 04: Rust Relationship Extraction & Comment Attachment

## Objective

Add Rust relationship extraction (`use` imports, function calls, `impl ... for ...` trait implementations, `mod` module references) and comment attachment (`///` outer doc, `//!` inner doc, `//` regular). Rust's unique patterns (impl blocks as containers, trait implementations) require the most custom logic.

## must_haves

- Rust files produce IMPORTS relationships for `use` declarations
- Rust files produce CALLS relationships for function calls
- Rust files produce INHERITS relationships for `impl Trait for Type` (semantic inheritance)
- Rust files produce INCLUDES relationships for `mod` declarations (module references)
- `///` outer doc comments attach to the following chunk
- `//!` inner doc comments attach to the enclosing module (best-effort: attach to first chunk)
- Regular `//` comments use proximity heuristic

```python
def _extract_rust_use(
    node: Node,
    source_chunks: list[Chunk],
    file_path: str,
) -> list[Relationship]:
    """Extract Rust use declarations as import relationships."""
    relationships: list[Relationship] = []
    source_id = source_chunks[0].id if source_chunks else ""

    def _walk(n: Node) -> None:
        if n.type == "use_declaration":
            source_line = n.start_point[0] + 1
            # Extract the use path from argument child
            for child in n.children:
                if child.type in ("scoped_identifier", "scoped_use_list", "use_list", "identifier"):
                    use_path = child.text.decode("utf-8").strip()
                    # Clean up: remove 'use ' prefix if present, trailing ';'
                    # tree-sitter typically gives us just the path part
                    relationships.append(Relationship(
                        source_id=source_id,
                        target_id="",
                        target_name=use_path,
                        rel_type=RelType.IMPORTS,
                        confidence=Confidence.INFERRED,
                        source_line=source_line,
                    ))
                    break
            else:
                # Fallback: extract text content between 'use' and ';'
                text = n.text.decode("utf-8").strip()
                if text.startswith("use "):
                    path = text[4:].rstrip(";").strip()
                    if path:
                        relationships.append(Relationship(
                            source_id=source_id,
                            target_id="",
                            target_name=path,
                            rel_type=RelType.IMPORTS,
                            confidence=Confidence.INFERRED,
                            source_line=source_line,
                        ))

        for child in n.children:
            _walk(child)

    _walk(node)
    return relationships
```

```python
def _extract_rust_calls(
    node: Node,
    source_chunks: list[Chunk],
    file_path: str,
) -> list[Relationship]:
    """Extract Rust function call relationships."""
    relationships: list[Relationship] = []

    def _walk(n: Node) -> None:
        if n.type == "call_expression":
            callee = n.children[0] if n.children else None
            if callee is None:
                for child in n.children:
                    _walk(child)
                return

            source_line = n.start_point[0] + 1
            enclosing = _find_enclosing_chunk(n, source_chunks)
            if enclosing is None:
                for child in n.children:
                    _walk(child)
                return

            # Extract callee name
            callee_name = callee.text.decode("utf-8")
            # Simplify scoped calls: std::collections::HashMap → HashMap
            simple_name = callee_name.split("::")[-1]
            # Handle method calls: self.x → x, obj.method() → method
            if "." in simple_name:
                simple_name = simple_name.split(".")[-1]

            # Check same-file chunks
            target_chunk = None
            for chunk in source_chunks:
                if chunk.name == simple_name:
                    target_chunk = chunk
                    break

            if target_chunk is not None:
                relationships.append(Relationship(
                    source_id=enclosing.id,
                    target_id=target_chunk.id,
                    target_name=callee_name,
                    rel_type=RelType.CALLS,
                    confidence=Confidence.DIRECT,
                    source_line=source_line,
                ))
            else:
                relationships.append(Relationship(
                    source_id=enclosing.id,
                    target_id="",
                    target_name=callee_name,
                    rel_type=RelType.CALLS,
                    confidence=Confidence.INFERRED,
                    source_line=source_line,
                ))

        for child in n.children:
            _walk(child)

    _walk(node)
    return relationships
```

```python
def _extract_rust_impl(
    node: Node,
    source_chunks: list[Chunk],
    file_path: str,
) -> list[Relationship]:
    """Extract Rust impl relationships (impl Trait for Type)."""
    relationships: list[Relationship] = []

    def _walk(n: Node) -> None:
        if n.type == "impl_item":
            # Look for "impl Trait for Type" pattern
            # tree-sitter-rust: impl_item has children including type_identifier and maybe 'for' keyword
            children = n.children
            impl_type = None
            trait_name = None
            has_for = False

            for i, child in enumerate(children):
                if child.type == "for":
                    has_for = True
                    # The identifier before 'for' is the trait name
                    if i > 0 and children[i-1].type in ("type_identifier", "identifier"):
                        trait_name = children[i-1].text.decode("utf-8")
                    # The identifier after 'for' is the type being implemented for
                    if i + 1 < len(children) and children[i+1].type in ("type_identifier", "identifier"):
                        impl_type = children[i+1].text.decode("utf-8")

            if trait_name and impl_type:
                # Find the type's chunk
                type_chunk = None
                for chunk in source_chunks:
                    if chunk.name == impl_type and chunk.chunk_type.value == "class":
                        type_chunk = chunk
                        break

                if type_chunk is not None:
                    relationships.append(Relationship(
                        source_id=type_chunk.id,
                        target_id="",
                        target_name=trait_name,
                        rel_type=RelType.INHERITS,
                        confidence=Confidence.INFERRED,
                        source_line=n.start_point[0] + 1,
                    ))

        for child in n.children:
            _walk(child)

    _walk(node)
    return relationships
```

```python
def _extract_rust_mod(
    node: Node,
    source_chunks: list[Chunk],
    file_path: str,
) -> list[Relationship]:
    """Extract Rust mod declarations as module reference relationships."""
    relationships: list[Relationship] = []
    source_id = source_chunks[0].id if source_chunks else ""

    def _walk(n: Node) -> None:
        if n.type == "mod_item":
            source_line = n.start_point[0] + 1
            name_node = n.child_by_field_name("name")
            if name_node:
                mod_name = name_node.text.decode("utf-8")
                relationships.append(Relationship(
                    source_id=source_id,
                    target_id="",
                    target_name=mod_name,
                    rel_type=RelType.INCLUDES,
                    confidence=Confidence.INFERRED,
                    source_line=source_line,
                ))

        for child in n.children:
            _walk(child)

    _walk(node)
    return relationships
```

```python
def extract_rust_relationships(
    filepath: Path,
    language: Language,
    repo_root: Path,
    source_chunks: list[Chunk],
    store: LadybugStore,
) -> list[Relationship]:
    """Extract all relationships from a Rust source file."""
    root = get_root_node(filepath, language)
    if root is None:
        return []

    relative_path = str(filepath.relative_to(repo_root))
    use_rels = _extract_rust_use(root, source_chunks, relative_path)
    calls = _extract_rust_calls(root, source_chunks, relative_path)
    impl_rels = _extract_rust_impl(root, source_chunks, relative_path)
    mod_rels = _extract_rust_mod(root, source_chunks, relative_path)

    return use_rels + calls + impl_rels + mod_rels
```

```python
if language == Language.RUST and root is not None:
    comment_types = COMMENT_TYPES.get(language, set())
    def _collect_inner_doc_comments(n: Node) -> list[str]:
        inner_docs = []
        for child in n.children:
            if child.type in comment_types:
                text = child.text.decode("utf-8")
                if text.startswith("//!"):
                    content = text[3:].strip()
                    inner_docs.append(content)
        return inner_docs

    inner_docs = _collect_inner_doc_comments(root)
    if inner_docs and chunks:
        # Attach to first top-level chunk (no parent_id)
        first_top = None
        for c in chunks:
            if c.parent_id is None:
                first_top = c
                break
        if first_top is None:
            first_top = chunks[0]
        for doc in inner_docs:
            if doc not in first_top.attached_comments:
                first_top.attached_comments.append(doc)
```

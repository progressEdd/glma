3. Implement `resolveModelHint()` function:
```typescript
function resolveModelHint(
    hint: string,
    availableModels: Array<{ id: string; provider: string; cost: { input: number; output: number } }>,
    currentModel: any,
): any | null {
    if (!hint || hint.trim() === "") return currentModel;

    if (hint === "fast") {
        const fast = availableModels.filter(m =>
            FAST_FAMILIES.some(f => m.id.toLowerCase().includes(f))
        );
        if (fast.length > 0) {
            return fast.sort((a, b) => (a.cost.input + a.cost.output) - (b.cost.input + b.cost.output))[0];
        }
        return availableModels.sort((a, b) => (a.cost.input + a.cost.output) - (b.cost.input + b.cost.output))[0] || null;
    }

    if (hint === "capable") {
        const capable = availableModels.filter(m =>
            CAPABLE_FAMILIES.some(f => m.id.toLowerCase().includes(f))
        );
        if (capable.length > 0) {
            return capable.sort((a, b) => (b.cost.input + b.cost.output) - (a.cost.input + a.cost.output))[0];
        }
        return availableModels.sort((a, b) => (b.cost.input + b.cost.output) - (a.cost.input + a.cost.output))[0] || null;
    }

    // Exact model ID — search across all providers
    return availableModels.find(m => m.id === hint) || null;
}
```

4. Implement `runGlmaSummarize()` function that:
   a. Reads `model_hint` from `.glma.toml` (using `node:fs` to read the file, parse with basic regex/toml parsing)
   b. Resolves the hint using `resolveModelHint()` against `ctx.modelRegistry.getAvailable()`
   c. Gets API key via `ctx.modelRegistry.getApiKeyAndHeaders(model)` for cloud models
   d. Determines if model is local or cloud:
      - If model's `baseUrl` contains `localhost` or matches a known local preset → local model
      - Otherwise → cloud model
   e. For local models: shells out to `glma index --summarize --summarize-provider <preset> --summarize-model <model>` using `child_process.execSync`
   f. For cloud models: reads chunks needing summaries by shelling out to a query command, calls `complete()` for each chunk, writes summaries back via `glma` write-back command
   
   Actually, per CONTEXT.md D-02 and the discussion log, simplify: 
   - For ALL models, the extension can shell out to `glma index --summarize` with the right flags
   - For cloud models, it sets the API key as an env var and passes the cloud provider's base_url
   - This reuses the entire Python pipeline (decomposition, incremental, error handling)

   Implementation:
```typescript
import { execSync } from "node:child_process";
import * as fs from "node:fs";
import * as path from "node:path";

async function runGlmaSummarize(hint: string, ctx: any): Promise<string> {
    const cwd = ctx.cwd;
    const available = ctx.modelRegistry.getAvailable();
    const model = resolveModelHint(hint, available, ctx.model);

    if (!model) {
        return `Error: Could not resolve model hint "${hint}". Available models: ${available.map((m: any) => m.id).join(", ")}`;
    }

    // Get auth for the model
    const auth = await ctx.modelRegistry.getApiKeyAndHeaders(model);
    if (!auth?.ok || !auth?.apiKey) {
        return `Error: No API key available for model ${model.provider}/${model.id}. Configure via /login or settings.`;
    }

    // Check if this is a local model (localhost URL or known preset)
    const isLocal = model.baseUrl.includes("localhost") || model.baseUrl.includes("127.0.0.1");

    const env = { ...process.env };
    let command: string;

    if (isLocal) {
        // Find matching preset for the base URL
        const preset = Object.entries(GLMA_PRESETS).find(([_, v]) => v.baseUrl === model.baseUrl);
        const providerFlag = preset ? `--summarize-provider ${preset[0]}` : `--summarize-provider local`;
        command = `glma index --summarize ${providerFlag} --summarize-model ${model.id}`;
    } else {
        // Cloud model — pass URL and API key via env
        env.OPENAI_API_KEY = auth.apiKey;
        if (auth.headers) {
            Object.assign(env, auth.headers);
        }
        command = `glma index --summarize --summarize-provider local --ai-url ${model.baseUrl} --summarize-model ${model.id}`;
    }

    try {
        const output = execSync(command, {
            cwd,
            env,
            timeout: 300000, // 5 min timeout for large repos
            encoding: "utf-8",
        });
        return output || "Summarization complete.";
    } catch (e: any) {
        return `Error during summarization: ${e.message}`;
    }
}
```

5. Register the `/glma-summarize` command:
```typescript
pi.registerCommand("glma-summarize", {
    description: "Summarize codebase chunks using AI via pi's model registry",
    handler: async (args, ctx) => {
        ctx.ui.notify("Starting glma summarization...", "info");

        // Read model_hint from .glma.toml
        const configPath = path.join(ctx.cwd, ".glma.toml");
        let hint = "";
        if (fs.existsSync(configPath)) {
            const content = fs.readFileSync(configPath, "utf-8");
            const match = content.match(/model_hint\s*=\s*["']([^"']+)["']/);
            if (match) hint = match[1];
        }

        const result = await runGlmaSummarize(hint, ctx);
        ctx.ui.notify(result, result.startsWith("Error") ? "error" : "success");
    },
});
```
